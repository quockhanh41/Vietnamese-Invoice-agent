# M5-C: the HTTP API — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Put an HTTP surface on the system that adds no logic of its own — every route a thin adapter over `submit_decision`, a repository, or `db/queries.py` — so that approving an invoice over `curl` exercises exactly the path a test and a script already exercise.

**Architecture:** `src/invoice_agent/api/` with one module per route group, sync `def` routes reusing the existing sync engine. `POST /batches` spools uploads, runs the batch on a background thread, and publishes per-node updates to an in-process queue that SSE drains; the queue is expected to be lost on restart, so `GET /batches/{id}` reads Postgres instead.

**Tech Stack:** FastAPI 0.140 · Starlette 1.3.1 · uvicorn 0.51 · `python-multipart` (**new pin**) · SQLAlchemy 2 Core (sync) · pytest + `fastapi.testclient.TestClient` (httpx 0.28.1 already present)

**Spec:** [`docs/superpowers/specs/2026-08-01-m5c-fastapi-design.md`](../specs/2026-08-01-m5c-fastapi-design.md)

## Global Constraints

Copied from spec §0. Every task's requirements implicitly include these.

1. **The API adds no logic, and no route contains SQL.** Every route adapts a function in the data or graph layer.
2. **The ledger has exactly one door.** No module under `api/routes/` may import `tools.guards` or `db.repositories.journal`.
3. **The request schema is the security boundary.** `DecisionRequest` uses `extra="forbid"` and carries no `posting_authority`, `approval_id`, `status` or `entry_id`.
4. **Progress is ephemeral; the database is truth.** The SSE registry dies on restart, by design §5.5.
5. **Sync all the way down.** Plain `def` routes. Design §2.10's `AsyncConnectionPool` is deliberately not implemented.
6. **No auth.** `decided_by` comes from the request body; §11.3 cut auth on purpose.
7. **Language:** code, comments, docstrings, commits in **English**. Product output stays **Vietnamese** — flag narratives and `name_vi` pass through the API unchanged and untranslated.
8. **Every non-obvious constant carries date + measurement + report filename.**
9. **`make check` after EVERY task**, and **push before claiming anything**. M5-B was declared COMPLETE with 12 commits unpushed and CI never run.

```python
"""Read-only list and aggregate queries — everything the HTTP API reads.

WHY THIS MODULE EXISTS. Half of M5-C's routes had nothing to wrap: the
repositories carry `get_by_id`, `get_by_natural_key`, `get_lines` and
`set_status`, but no list, no filter and no counts. The alternative was SQL
inside route modules, which makes "the API adds no logic" false on the first
commit and strands aggregate SQL where no non-HTTP caller can reuse it —
M5-D's Streamlit and M8's ReportingAgent both want these same numbers.

READ-ONLY IS A RULE, NOT A DESCRIPTION. Nothing here writes.
`tests/test_api_boundary.py::test_queries_module_is_read_only` enforces it, so
this cannot grow into a second write path beside the repositories.
"""

import uuid
from typing import Any

from sqlalchemy import Connection, func, select

from invoice_agent.db import models


def list_invoices(
    conn: Connection,
    *,
    status: str | None = None,
    department: str | None = None,
    limit: int = 50,
) -> list[dict[str, Any]]:
    inv = models.invoices
    stmt = select(
        inv.c.id, inv.c.invoice_series, inv.c.invoice_number, inv.c.seller_tax_code,
        inv.c.issue_date, inv.c.total_amount, inv.c.category, inv.c.department,
        inv.c.status, inv.c.batch_id,
    )
    if status is not None:
        stmt = stmt.where(inv.c.status == status)
    if department is not None:
        stmt = stmt.where(inv.c.department == department)
    stmt = stmt.order_by(inv.c.created_at.desc()).limit(limit)
    return [dict(r) for r in conn.execute(stmt).mappings()]


def batch_summary(conn: Connection, batch_id: uuid.UUID) -> dict[str, Any] | None:
    """Every number is a COUNT against a table, never the progress registry.

    Spec §2.4: the registry is in process memory and is expected to be lost.
    This answers correctly whether or not anyone was ever streaming.
    """
    row = (
        conn.execute(select(models.batches).where(models.batches.c.id == batch_id))
        .mappings()
        .first()
    )
    if row is None:
        return None

    inv = models.invoices

    def _count(*where: Any) -> int:
        return int(
            conn.execute(select(func.count()).select_from(inv).where(*where)).scalar_one()
        )

    pending = int(
        conn.execute(
            select(func.count())
            .select_from(models.approvals.join(inv, models.approvals.c.invoice_id == inv.c.id))
            .where(inv.c.batch_id == batch_id, models.approvals.c.decision.is_(None))
        ).scalar_one()
    )
    return {
        "batch_id": str(batch_id),
        "status": row["status"],
        "invoice_count": row["invoice_count"],
        "persisted": _count(inv.c.batch_id == batch_id),
        "posted": _count(inv.c.batch_id == batch_id, inv.c.status == "posted"),
        "flagged": _count(inv.c.batch_id == batch_id, inv.c.status == "flagged"),
        "pending_approval": pending,
    }


def budget_report(conn: Connection, *, period: str | None = None) -> list[dict[str, Any]]:
    b, bc = models.budgets, models.budget_consumption
    consumed = (
        select(bc.c.budget_id, func.coalesce(func.sum(bc.c.amount), 0).label("consumed"))
        .group_by(bc.c.budget_id)
        .subquery()
    )
    stmt = (
        select(
            b.c.department_code, b.c.period, b.c.category, b.c.amount_limit,
            b.c.warn_threshold_pct,
            func.coalesce(consumed.c.consumed, 0).label("consumed"),
        )
        .select_from(b.outerjoin(consumed, consumed.c.budget_id == b.c.id))
        .order_by(b.c.department_code, b.c.period)
    )
    if period is not None:
        stmt = stmt.where(b.c.period == period)
    rows = [dict(r) for r in conn.execute(stmt).mappings()]
    for r in rows:
        # Stringified because Decimal is not JSON-serialisable and this goes
        # straight out over HTTP. Converting at the boundary, as the approval
        # graph's `_jsonable` does, means one place to look.
        r["amount_limit"] = str(r["amount_limit"])
        r["warn_threshold_pct"] = str(r["warn_threshold_pct"])
        r["consumed"] = str(r["consumed"])
        r["remaining"] = str(r["amount_limit"])  # replaced below, kept typed
    for r in rows:
        from decimal import Decimal as _D

        r["remaining"] = str(_D(r["amount_limit"]) - _D(r["consumed"]))
    return rows


def flag_report(conn: Connection) -> list[dict[str, Any]]:
    f = models.invoice_flags
    stmt = (
        select(
            f.c.code, f.c.severity,
            func.count().label("total"),
            func.count().filter(f.c.resolved_at.is_(None)).label("open_count"),
        )
        .group_by(f.c.code, f.c.severity)
        .order_by(func.count().desc())
    )
    return [dict(r) for r in conn.execute(stmt).mappings()]


def posting_report(conn: Connection) -> list[dict[str, Any]]:
    j = models.journal_entries
    stmt = (
        select(j.c.posting_authority, func.count().label("entries"))
        .group_by(j.c.posting_authority)
        .order_by(j.c.posting_authority)
    )
    return [dict(r) for r in conn.execute(stmt).mappings()]


def cost_report(conn: Connection) -> dict[str, Any]:
    """Real AND shadow. On Groq free tier `cost_usd` is $0, which reads as
    "you did not measure" rather than "it is cheap" — design §2.9."""
    c = models.llm_calls
    row = conn.execute(
        select(
            func.count().label("calls"),
            func.count().filter(c.c.succeeded).label("succeeded"),
            func.coalesce(func.sum(c.c.cost_usd), 0).label("cost_usd"),
            func.coalesce(func.sum(c.c.shadow_cost_usd), 0).label("shadow_cost_usd"),
        ).select_from(c)
    ).mappings().one()
    return {
        "calls": int(row["calls"]),
        "succeeded": int(row["succeeded"]),
        "cost_usd": str(row["cost_usd"]),
        "shadow_cost_usd": str(row["shadow_cost_usd"]),
    }
```

"""Liveness that actually checks the thing it claims to check."""

from typing import Any

from fastapi import APIRouter
from fastapi.responses import JSONResponse
from sqlalchemy import text

from invoice_agent.db.engine import get_engine

router = APIRouter(tags=["health"])


@router.get("/health")
def health() -> Any:
    """PINGS POSTGRES. A health check that does not touch the database reports
    healthy while the database is gone — worse than no health check, because a
    deploy pipeline believes it.

    `get_engine` is module-level so a test can replace it; that is also how the
    unreachable branch is exercised without stopping the container.
    """
    try:
        with get_engine().connect() as conn:
            conn.execute(text("SELECT 1"))
    except Exception:
        return JSONResponse(
            status_code=503, content={"status": "degraded", "database": "unreachable"}
        )
    return {"status": "ok", "database": "ok"}

```python
"""The FastAPI application — routes split for real (design §11.2).

Project A's API was one 643-line module beside an empty `routes/` package. This
one is split from the first commit, one module per route group.

NO GLOBAL `ValueError` HANDLER, deliberately. It would catch genuine bugs and
serve them as 422 client errors — the API equivalent of a bare `except`. M5-B's
review found a bare `assert` in `serial_join` that killed a whole batch and was
invisible under replay; a broad handler here is the same mistake with a
friendlier status code. Routes raise `HTTPException` at the call site instead.
"""

from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

from invoice_agent.api.routes import health
from invoice_agent.tools.guards import GuardDenied


def install_error_handlers(app: FastAPI) -> None:
    @app.exception_handler(GuardDenied)
    def _guard_denied(_request: Request, exc: GuardDenied) -> JSONResponse:
        # 409, not 500. The guard refusing is the system working; a 500 would
        # read as a bug and bury a working safety control in the logs. The
        # guard has already written its own `audit_log` row on its own
        # connection, so the refusal is durable regardless of this response.
        return JSONResponse(status_code=409, content={"detail": str(exc)})


def create_app() -> FastAPI:
    app = FastAPI(
        title="Vietnamese Invoice & Expense Agent",
        version="0.1.0",
    )
    install_error_handlers(app)
    app.include_router(health.router)
    return app


app = create_app()
```

def test_the_budget_report_sends_money_as_strings(committed_rows):
    """Amounts cross the wire as STRINGS, and the reason is precise.

    MEASURED 2026-08-01: `fastapi.encoders.jsonable_encoder` turns `Decimal`
    into `float`. For realistic VND amounts the round-trip is exact — even
    21618409.09000000, the widest real fixture value, survives — so the claim is
    NOT "a float loses the last đồng" at these magnitudes. Loss appears past ~17
    significant digits (12345678901234567.89 -> 1.2345678901234568e+16).

    Two things are lost that do matter here. SCALE: 21618409.09000000 becomes
    21618409.09, discarding what the source declared. And ARITHMETIC: a client
    that sums float amounts inherits binary-float semantics, so a dashboard
    total can disagree with `budget_consumption.balance_after`, which Postgres
    computed in NUMERIC. This system chose Decimal at M1 for exactly that; the
    wire format should not quietly undo it."""

def _jsonable(value: Any) -> Any:
    """Decimal, date and UUID to strings at the boundary.

    Money is `Decimal` by a decision made at M1. Measured 2026-08-01:
    `jsonable_encoder` turns Decimal into float, which round-trips exactly at
    invoice magnitudes but discards SCALE (21618409.09000000 -> 21618409.09) and
    hands clients float arithmetic, so a summed dashboard total can disagree
    with a NUMERIC `balance_after`.
    """
    import datetime
    from decimal import Decimal

    if isinstance(value, Decimal | uuid.UUID):
        return str(value)
    if isinstance(value, datetime.date | datetime.datetime):
        return value.isoformat()
    if isinstance(value, dict):
        return {k: _jsonable(v) for k, v in value.items()}
    if isinstance(value, list):
        return [_jsonable(v) for v in value]
    return value

"""Request models — the security boundary at the HTTP edge.

Design §9.2 makes this argument for agent response models against prompt
injection; it transposes directly to clients. The defence is not validation, it
is that the dangerous instruction IS NOT EXPRESSIBLE: `DecisionRequest` has no
`posting_authority`, no `approval_id`, no `status` and no `entry_id`, so no
request body can carry them.

`extra="forbid"` is the other half. Without it an unknown field is silently
ignored, and an audit of the field list would be checking a schema the request
is not actually held to.
"""

from typing import Any, Literal

from pydantic import BaseModel, ConfigDict, Field


class DecisionRequest(BaseModel):
    model_config = ConfigDict(extra="forbid")

    decision: Literal["approved", "rejected", "edited"]
    # BARE name. `approvals.decided_by` stores "ketoan"; `audit_log.actor` gets
    # the `human:` prefix, and `post_entry` is the only place it is added.
    #
    # NO AUTH, deliberately: design §11.3 cut it, and §4 defines the reviewer as
    # a string in `approvals.decided_by`. This records what the client CLAIMS. A
    # real deployment would take it from a session.
    decided_by: str = Field(min_length=1)
    reason: str | None = None
    corrections: dict[str, Any] | None = None

```python
"""In-process progress channels for running batches.

EXPECTED TO BE LOST. This is memory, not storage: a restart empties it, and
design §5.5 says the API never assumes an in-RAM registry survived one. That is
why `GET /batches/{id}` counts rows in Postgres instead of reading this, and why
`tests/test_api_batches.py` clears the registry and asserts the batch is still
queryable.

Losing progress costs a client its live view. It never costs a batch.
"""

import queue
import threading
from collections.abc import Iterator
from typing import Any

_CHANNELS: dict[str, queue.Queue[dict[str, Any] | None]] = {}
_LOCK = threading.Lock()

# Sentinel pushed by `finish` so a streaming client terminates instead of
# blocking forever on a batch that already ended.
_END = None


def _channel(batch_id: str) -> queue.Queue[dict[str, Any] | None]:
    with _LOCK:
        return _CHANNELS.setdefault(batch_id, queue.Queue())


def publish(batch_id: str, event: dict[str, Any]) -> None:
    _channel(batch_id).put(event)


def finish(batch_id: str) -> None:
    _channel(batch_id).put(_END)


def stream(batch_id: str, *, timeout: float = 120.0) -> Iterator[dict[str, Any]]:
    """Yield events until the batch finishes or `timeout` elapses.

    The timeout is a liveness guard, not a deadline for the batch: a client that
    attaches to a batch whose producer already died would otherwise hang for as
    long as the connection stayed open.
    """
    channel = _channel(batch_id)
    while True:
        try:
            event = channel.get(timeout=timeout)
        except queue.Empty:
            return
        if event is _END:
            return
        yield event


def clear() -> None:
    """Drop every channel. Stands in for a process restart in tests."""
    with _LOCK:
        _CHANNELS.clear()
```

        "persisted": [], "consumption": [], "raised": [], "unroutable": [],

    }
    try:
        for chunk in build_batch_graph().stream(state, stream_mode="updates"):
            for node, update in chunk.items():
                progress.publish(
                    str(batch_id),
                    {"node": node, "status": (update or {}).get("status")},
                )
    except Exception as exc:  # noqa: BLE001 - a failed batch must be reportable
        # The failure has to reach the DATABASE, not just this thread's stack.
        # A client polling GET /batches/{id} is the only observer that survives
        # the thread, so `batches.status` is where the truth goes.
        with get_engine().begin() as conn:
            batches_repo.complete_batch(conn, batch_id, status="failed", invoice_count=0)
        progress.publish(str(batch_id), {"node": "error", "detail": str(exc)})
    finally:
        progress.publish(str(batch_id), {"event": "end"})
        progress.finish(str(batch_id))


@router.post("", status_code=202)
def create_batch(
    files: list[UploadFile] = File(...),
    department: str | None = Form(default=None),
) -> dict[str, Any]:
    """`department` is OPTIONAL and its absence has a consequence worth stating.

    `serial_join` groups budgets by the department declared on the TASK;
    `InvoiceData` carries no department, so an upload without one is persisted
    and may be posted, but lands in `unroutable` and is never reconciled against
    a budget. Recorded in the spec's open questions rather than papered over
    with a default — M4-D deleted an invented budget limit for the same reason.
    """
    if not files:
        raise HTTPException(status_code=422, detail="at least one file is required")

    batch_id = uuid.uuid4()
    inputs = _spool(batch_id, files)
    with get_engine().begin() as conn:
        batches_repo.ensure_batch(conn, batch_id, label=f"upload:{len(inputs)}")

    threading.Thread(target=_run, args=(batch_id, inputs, department), daemon=True).start()
    return {"batch_id": str(batch_id), "invoice_count": len(inputs)}


@router.get("/{batch_id}")
def get_batch(batch_id: uuid.UUID) -> dict[str, Any]:
    with get_engine().begin() as conn:
        summary = queries.batch_summary(conn, batch_id)
    if summary is None:
        raise HTTPException(status_code=404, detail=f"batch {batch_id} not found")
    return summary


@router.get("/{batch_id}/events")
def events(batch_id: uuid.UUID) -> StreamingResponse:
    import json

    def _generate():
        for event in progress.stream(str(batch_id)):
            yield f"data: {json.dumps(event, ensure_ascii=False)}\n\n"

    return StreamingResponse(_generate(), media_type="text/event-stream")

import pathlib

def test_no_route_reaches_the_ledger_directly():
    """THE LEDGER HAS ONE DOOR. Posting reaches `journal_entries` only through
    submit_decision -> post_entry -> the guard.

    Extends `test_only_the_guard_may_call_the_journal_repository` (M5-A), which
    allowlists guards.py as the sole caller of the journal repository. This stops
    a ROUTE from becoming the second caller by importing the guard and choosing
    its own posting_authority — a bypass no behavioural test would notice,
    because a correctly-posted entry looks the same either way.

    offenders = []
    for path in ROUTES.rglob("*.py"):
        source = path.read_text(encoding="utf-8")
        if "tools.guards" in source or "tools import guards" in source:
            offenders.append(f"{path}: imports tools.guards")
        if "repositories.journal" in source or "repositories import journal" in source:
            offenders.append(f"{path}: imports the journal repository")
    assert not offenders, (
        "the ledger must be reached through submit_decision only: " + "; ".join(offenders)
    )

    """`db/queries.py` exists to serve reads. A write here would be a second
